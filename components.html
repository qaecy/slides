<cue-logo>

Animated QAECY logo — adapts to light and dark mode

<cue-logo> — API

Attribute Type Default Description
size 'xs' | 's' | 'm' | 'l' 'm' Size preset
active boolean true Plays the draw animation once on load. Set to false to skip it
continuous boolean false Loops the draw animation indefinitely
mode 'default' | 'glow' 'default' 'glow' — only the teal tip traverses the path in an infinite loop. active and continuous are ignored in this mode

<cue-logo> — size presets

size="s"
size="m" (default)
size="l"

<cue-logo> — animation

↺ refresh to replay

active="false" — outlines once
active (default) — draws once
continuous — loops

<cue-logo> — glow mode

mode="glow" shows only the teal tip traversing the path — ideal as a subtle ambient loader. active and continuous are ignored.

mode="default" continuous
mode="glow"

<cue-logo> — dark & light mode

Colours adapt when class="dark" is present on <body>

⬤ toggle dark / light

<cue-logo> — dark & light mode 

<!-- Dark mode -->
<body class="dark">
  <cue-logo size="m"></cue-logo>
</body>

<!-- Light mode -->
<body>
  <cue-logo size="m"></cue-logo>
</body>

<!-- Toggle at runtime -->
<script>
  document.body.classList.toggle('dark');
</script>

<cue-logo> — usage 

<!-- Outline draw and out -->
<cue-logo size="l" active="false"></cue-logo>

<!-- Animated once on load (default behaviour) -->
<cue-logo size="m"></cue-logo>

<!-- Continuously looping -->
<cue-logo size="m" continuous></cue-logo>

<!-- Glow mode — teal tip loops infinitely, no draw stroke -->
<cue-logo size="m" mode="glow"></cue-logo>

<!-- Inline (scale to text size with CSS) -->
<style>
  .logo-inline { display: inline-block; height: 1em; vertical-align: middle; }
</style>
<cue-logo class="logo-inline" size="s"></cue-logo>

<cue-document-viewer>

In-browser viewer for PDF, BIM, CAD, images, spreadsheets, and more

<cue-document-viewer> — API

Property Type Default Description
cue Cue — Authenticated Cue SDK instance. Set via JS property after sign-in
uuid string — Document UUID. Optionally append ?page=N to open at a specific page
projectId string — Cue project ID the document belongs to
showDocumentInfo boolean true Shows a document info header with name, download, and context menu
variant 'none' | 'default' | … 'none' Card wrapper style (same values as <cue-document-list>)
portalBaseURL string 'https://cue.qaecy.com' Base URL used for the "Open in Cue" context-menu link

<cue-document-viewer> — variant

Wraps the viewer in a styled card. Defaults to none (borderless).

Value Description
none No wrapper — component fills its container directly
default Neutral card with border and subtle background
primary Card with primary brand colour accent
secondary Card with secondary brand colour accent
accent Card with accent colour highlight
fade Soft, low-contrast card
variant=

<cue-document-viewer> — usage

<!-- 1. Include the cue-ui loader -->
<script src="https://cdn.jsdelivr.net/npm/@qaecy/cue-ui@{{CUE_UI_VERSION}}/index.js"
  type="module" data-components="cue-document-viewer"></script>

<!-- 2. Add the viewer element -->
<cue-document-viewer-wc id="dv" style="display:block; height:600px;"></cue-document-viewer-wc>

<!-- 3. Wire up after auth -->
<script type="module">
  import { Cue } from 'https://esm.sh/@qaecy/cue-sdk@{{CUE_SDK_VERSION}}';

  const cue = new Cue();
  await cue.auth.signIn('google');

  await customElements.whenDefined('cue-document-viewer-wc');
  const dv = document.getElementById('dv');
  dv.cue       = cue;
  dv.projectId = 'my-project-id';
  dv.uuid      = 'document-uuid';
</script>

<cue-document-list>

Paginated, sortable table of Cue project documents

<cue-document-list> — API

Attribute / Property Type Default Description

<cue-document-list> — page-size

Controls how many rows are shown per page.

Attribute Type Default Description
page-size number 10 Rows visible on each page. Smaller values show the paginator sooner; larger values reduce page-turn frequency for dense datasets.
prefetch-pages number 3 Pages fetched ahead in the background so navigation feels instant.
page-size=

<cue-document-list> — usage 

<!-- 1. Include the cue-ui loader -->
<script src="https://cdn.jsdelivr.net/npm/@qaecy/cue-ui@{{CUE_UI_VERSION}}/index.js" type="module"></script>

<!-- 2. Add the toolbar (handles auth + project select) -->
<script type="module" src="js/components/cue-sdk-toolbar.js"></script>
<cue-sdk-toolbar id="tb"></cue-sdk-toolbar>

<!-- 3. Add the document list -->
<cue-document-list id="dl" page-size="25"></cue-document-list>

<!-- 4. Wire them up in JS -->
<script type="module">
  const QCY = 'https://dev.qaecy.com/ont#';
  const tb = document.getElementById('tb');
  const dl = document.getElementById('dl');

  tb.addEventListener('projectchange', async ({ detail: { projectId, sdkState } }) => {
    dl.projectId = projectId;
    dl.sdkState  = sdkState;

    if (projectId && sdkState) {
      // Optional: filter by file type via SPARQL
      const data = await sdkState.cue.api.sparql(
        `SELECT ?id WHERE { ?fc a <${QCY}FileContent> }
         BIND(REPLACE(STR(?fc),"^.*/([^/]*)$","$1") AS ?id)`,
        projectId
      );
      dl.uuids = data.results.bindings.map(b => b.id.value);
    }
  });
</script>

secondary Card with secondary brand colour accent

<cue-entity-list> — usage

<!-- 1. Include the cue-ui loader -->
<script src="https://cdn.jsdelivr.net/npm/@qaecy/cue-ui@{{CUE_UI_VERSION}}/index.js"
  type="module" data-components="cue-entity-list"></script>

<!-- 2. Add the list element -->
<cue-ent-list id="el" style="display:block; height:600px;"></cue-ent-list>

<!-- 3. Wire up after auth -->
<script type="module">
  import { Cue, CueProjectEntities } from 'https://esm.sh/@qaecy/cue-sdk@{{CUE_SDK_VERSION}}';

  const cue = new Cue();
  await cue.auth.signIn('google');

  const projectId = 'my-project-id';
  const entities  = new CueProjectEntities(cue.api, projectId);

  await customElements.whenDefined('cue-ent-list');
  const el = document.getElementById('el');
  el.sdkState = { cue, entities };

  // Set entity IRIs to display (e.g. from a SPARQL query)
  const data = await cue.api.sparql(
    `SELECT ?e WHERE { ?e a <https://dev.qaecy.com/ont#CanonicalEntity> } LIMIT 50`,
    projectId
  );
  el.iris = data.results.bindings.map(b => b.e.value);
</script>
